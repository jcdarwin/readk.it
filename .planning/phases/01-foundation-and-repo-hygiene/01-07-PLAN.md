       ```sh
       #!/usr/bin/env sh
       # Pre-commit hook (Husky 9; modern flow — no v8 .husky.sh shim).
       # Defends Phase 1 hygiene gates:
       #   1. Reject any staged file > 1 MB (CONTEXT D-09 / Phase 1 BUILD-08 + Pitfall 10)
       #   2. Run lint-staged so ESLint + Prettier fix changed files only (CONTEXT D-10)

       set -e

       LIMIT=$((1024 * 1024))

       # NUL-delimited stage list to handle filenames with spaces / newlines (Pitfall 6).
       OVERSIZE_FILES=""
       while IFS= read -r -d '' f; do
         [ -z "$f" ] && continue
         [ -f "$f" ] || continue
         size=$(wc -c < "$f" | tr -d ' ')
         if [ "$size" -gt "$LIMIT" ]; then
           OVERSIZE_FILES="$OVERSIZE_FILES\n  - $f ($size bytes)"
         fi
       done < <(git diff --cached --name-only --diff-filter=ACMR -z)

       if [ -n "$OVERSIZE_FILES" ]; then
         printf 'ERROR: pre-commit rejected staged files larger than 1 MB:%b\n' "$OVERSIZE_FILES" >&2
         printf '\nUnstage with `git restore --staged <file>` and commit a smaller artefact.\n' >&2
         printf 'If a large binary is genuinely needed, add it to .gitignore or use Git LFS — but check `.planning/codebase/CONCERNS.md` first to confirm this is not the 700 MB-dist regression we are defending against.\n' >&2
         exit 1
       fi

       # Lint-staged runs ESLint + Prettier on staged TS/JS/JSON/MD/SCSS/CSS/HTML.
       # Configured inline in package.json (key: lint-staged).
       npx lint-staged
       ```
